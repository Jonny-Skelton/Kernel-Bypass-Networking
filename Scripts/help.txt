# 1) Ensure namespaces + veth exist
sudo ip netns add srv  || true
sudo ip netns add cli  || true
sudo ip link add veth0 type veth peer name veth1 || true
sudo ip link set veth0 netns srv
sudo ip link set veth1 netns cli
sudo ip netns exec srv  ip addr add 10.0.0.1/24 dev veth0
sudo ip netns exec srv  ip link set veth0 up
sudo ip netns exec cli  ip addr add 10.0.0.2/24 dev veth1
sudo ip netns exec cli  ip link set veth1 up

# 2) Clear any attached XDP
sudo ip netns exec srv  ip link set dev veth0 xdp off || true
sudo ip netns exec cli  ip link set dev veth1 xdp off || true

# 3) Attach pass-through on client (veth1)
sudo ip netns exec cli \
  ip link set dev veth1 xdp object ~/xdp-tutorial/basic01-xdp-pass/xdp_pass_kern.o sec xdp

# 4) Attach echo-TX on server (veth0)
sudo ip netns exec srv \
  ip link set dev veth0 xdp object ~/Kernel-Bypass-Networking/src/xdp/xdp_icmp_echo.o sec xdp

# 5) Confirm
sudo ip netns exec srv  bpftool net list
sudo ip netns exec cli  bpftool net list

# 6) Ping test
sudo ip netns exec cli ping -c4 10.0.0.1
